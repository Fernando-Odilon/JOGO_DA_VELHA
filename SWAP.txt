  const calcularJogadaHumana = (squares) => {
  const lines = [
    [0, 1, 2], [3, 4, 5], [6, 7, 8],
    [0, 3, 6], [1, 4, 7], [2, 5, 8],
    [0, 4, 8], [2, 4, 6]
  ];

  const posicoesDisponiveis = squares.map((val, idx) => (val === null ? idx : null)).filter((val) => val !== null);

  if (posicoesDisponiveis.length === 0) return null;

  // Função auxiliar para encontrar jogada de vitória ou bloqueio imediato
  const encontrarJogadaCritica = (jogador) => {
    for (let i = 0; i < lines.length; i++) {
      const [a, b, c] = lines[i];
      const linha = [squares[a], squares[b], squares[c]];
      const vaziosNaLinha = linha.filter((cell) => cell === null).length;
      const pecasDoJogador = linha.filter((cell) => cell === jogador).length;

      // Se há 2 peças do jogador e 1 espaço vazio, encontramos o ponto crítico
      if (pecasDoJogador === 2 && vaziosNaLinha === 1) {
        if (squares[a] === null) return a;
        if (squares[b] === null) return b;
        if (squares[c] === null) return c;
      }
    }
    return null;
  };

  // 1. Tenta vencer se tiver 2 em linha (chance de 85% de acerto para não parecer robótico)
  if (Math.random() < 0.85) {
    const jogadaVitoria = encontrarJogadaCritica('O');
    if (jogadaVitoria !== null) return jogadaVitoria;
  }

  // 2. Tenta bloquear o oponente se ele tiver 2 em linha (chance de 80% de bloqueio)
  if (Math.random() < 0.80) {
    const jogadaBloqueio = encontrarJogadaCritica('X');
    if (jogadaBloqueio !== null) return jogadaBloqueio;
  }

  // 3. Ocasionalmente tenta pegar o centro se estiver livre (comportamento natural)
  if (squares[4] === null && Math.random() < 0.60) {
    return 4;
  }

  // 4. Caso contrário, joga aleatoriamente entre as opções disponíveis
  const indiceAleatorio = Math.floor(Math.random() * posicoesDisponiveis.length);
  return posicoesDisponiveis[indiceAleatorio];
};

const getResponse = async (currentSquares, setPensando, setResposta) => {

  // Simula um tempo de reflexão humano (entre 400ms e 900ms)

  return null;
